Here’s a simple, standalone Python lead tracker. It saves to a CSV file, so you don’t need a database or any extra packages. It makes sure every lead has a next follow-up date, shows what’s overdue, and warns you about duplicate leads.

python
import csv
import os
from datetime import date, datetime, timedelta

FILE = "leads.csv"
FIELDS = ["id", "name", "phone", "email", "product", "source", "status",
          "priority", "created", "last_contact", "next_follow_up", "notes"]
STATUSES = ["New", "Contacted", "Quoted", "Follow-up", "Won", "Lost"]
CLOSED = {"Won", "Lost"}
PRIORITY_ORDER = {"high": 0, "medium": 1, "low": 2}


# ---------- storage ----------

def load_leads():
    if not os.path.exists(FILE):
        return []
    with open(FILE, newline="", encoding="utf-8") as f:
        return list(csv.DictReader(f))


def save_leads(leads):
    with open(FILE, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=FIELDS)
        writer.writeheader()
        writer.writerows(leads)


def next_id(leads):
    return str(max((int(l["id"]) for l in leads), default=0) + 1)


def find_lead(leads, lead_id):
    for lead in leads:
        if lead["id"] == lead_id.strip():
            return lead
    print("No lead with that ID.")
    return None


def parse_date(value):
    return datetime.strptime(value, "%Y-%m-%d").date() if value else None


# ---------- actions ----------

def add_lead(leads):
    name = input("Name: ").strip()
    if not name:
        print("Name is required.")
        return
    phone = input("Phone: ").strip()
    email = input("Email: ").strip()

    # Duplicate check so the same person isn't entered twice
    for l in leads:
        if (phone and l["phone"] == phone) or (email and l["email"].lower() == email.lower()):
            print(f"Warning: possible duplicate of lead #{l['id']} ({l['name']}, {l['status']}).")
            if input("Add anyway? (y/n): ").lower() != "y":
                return

    lead = {
        "id": next_id(leads),
        "name": name,
        "phone": phone,
        "email": email,
        "product": input("Product (auto/home/life/health/commercial): ").strip().lower(),
        "source": input("Source (referral/web/walk-in/etc): ").strip(),
        "status": "New",
        "priority": (input("Priority (high/medium/low) [medium]: ").strip().lower() or "medium"),
        "created": str(date.today()),
        "last_contact": "",
        "next_follow_up": str(date.today() + timedelta(days=1)),  # follow up tomorrow by default
        "notes": input("Notes: ").strip(),
    }
    leads.append(lead)
    save_leads(leads)
    print(f"Added lead #{lead['id']}. First follow-up: {lead['next_follow_up']}")


def log_contact(leads):
    lead = find_lead(leads, input("Lead ID: "))
    if not lead:
        return
    note = input("What happened on this contact? ").strip()
    days = input("Days until next follow-up [3]: ").strip()
    days = int(days) if days.isdigit() else 3

    today = date.today()
    lead["last_contact"] = str(today)
    lead["next_follow_up"] = str(today + timedelta(days=days))
    if note:
        stamp = f"{today}: {note}"
        lead["notes"] = f"{lead['notes']} | {stamp}".strip(" |")
    if lead["status"] == "New":
        lead["status"] = "Contacted"
    save_leads(leads)
    print(f"Logged. Next follow-up: {lead['next_follow_up']}")


def update_status(leads):
    lead = find_lead(leads, input("Lead ID: "))
    if not lead:
        return
    print("Statuses:", ", ".join(STATUSES))
    new_status = input("New status: ").strip().title()
    if new_status not in STATUSES:
        print("Unknown status.")
        return
    lead["status"] = new_status
    if new_status in CLOSED:
        lead["next_follow_up"] = ""  # closed leads don't need reminders
    elif not lead["next_follow_up"]:
        lead["next_follow_up"] = str(date.today() + timedelta(days=1))
    save_leads(leads)
    print("Status updated.")


def show_due(leads):
    today = date.today()
    open_leads = [l for l in leads if l["status"] not in CLOSED]
    due = []
    for l in open_leads:
        fu = parse_date(l["next_follow_up"])
        if fu is None or fu <= today:
            due.append((fu or today, l))

    if not due:
        print("Nothing due. Nice work.")
        return

    due.sort(key=lambda x: (x[0], PRIORITY_ORDER.get(x[1]["priority"], 1)))
    print(f"\n{'ID':>4}  {'Name':<22} {'Status':<10} {'Priority':<8} {'Follow-up':<11} Flag")
    print("-" * 75)
    for fu, l in due:
        if fu < today:
            flag = f"OVERDUE {(today - fu).days}d"
        elif fu == today:
            flag = "TODAY"
        else:
            flag = "NO DATE SET"
        print(f"{l['id']:>4}  {l['name'][:22]:<22} {l['status']:<10} {l['priority']:<8} "
              f"{str(fu):<11} {flag}")
    print()


def list_leads(leads):
    if not leads:
        print("No leads yet.")
        return
    print(f"\n{'ID':>4}  {'Name':<22} {'Product':<10} {'Status':<10} {'Priority':<8} Next follow-up")
    print("-" * 80)
    for l in sorted(leads, key=lambda x: int(x["id"])):
        print(f"{l['id']:>4}  {l['name'][:22]:<22} {l['product'][:10]:<10} {l['status']:<10} "
              f"{l['priority']:<8} {l['next_follow_up'] or '-'}")
    print()


def search(leads):
    term = input("Search name, phone, email, or notes: ").strip().lower()
    hits = [l for l in leads if any(term in l[k].lower() for k in ("name", "phone", "email", "notes"))]
    list_leads(hits) if hits else print("No matches.")


def show_stats(leads):
    today = date.today()
    total = len(leads)
    if total == 0:
        print("No leads yet.")
        return
    print(f"\nTotal leads: {total}")
    for s in STATUSES:
        print(f"  {s:<10} {sum(1 for l in leads if l['status'] == s)}")
    won = sum(1 for l in leads if l["status"] == "Won")
    lost = sum(1 for l in leads if l["status"] == "Lost")
    if won + lost:
        print(f"Win rate (closed leads): {won / (won + lost):.0%}")
    no_contact = sum(1 for l in leads if l["status"] not in CLOSED and not l["last_contact"])
    print(f"Open leads never contacted: {no_contact}\n")


# ---------- main menu ----------

MENU = """
1) Add lead
2) Log contact / call / email
3) Update status
4) Show due & overdue follow-ups
5) List all leads
6) Search
7) Stats
0) Quit
"""

def main():
    leads = load_leads()
    actions = {
        "1": add_lead, "2": log_contact, "3": update_status,
        "4": show_due, "5": list_leads, "6": search, "7": show_stats,
    }
    while True:
        print(MENU)
        choice = input("Choose: ").strip()
        if choice == "0":
            print("Goodbye.")
            break
        action = actions.get(choice)
        if action:
            action(leads)
        else:
            print("Invalid choice.")


if __name__ == "__main__":
    main()

How to use it

  1. Save the code as lead_keeper.py. Run python lead_keeper.py. It creates leads.csv in the same folder.

  2. Add every lead the moment you get it, even if you can’t reach them yet. Each new lead gets a follow-up set for tomorrow.

  3. Every morning, choose option 4. It lists everything overdue or due today, with high-priority leads first.

  4. After each call, email, or meeting, use option 2 to log what happened. This sets the next follow-up date, so the lead doesn’t fall off your list.

  5. Use option 3 to move leads through the pipeline. Closing a lead as Won or Lost removes its reminders.

Ways to customize it

  • Change STATUSES or the product and source options to match your agency’s process.

  • Add an “Renewal date” field for policy renewals, so you can follow up before the policy lapses.

  • Add a “Premium estimate” field to see the potential value of your pipeline.
